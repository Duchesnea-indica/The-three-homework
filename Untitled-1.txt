/* USER CODE BEGIN 4 */
void Led_Apply(void)
{
    // 先全部熄灭
    HAL_GPIO_WritePin(LED0_GPIO_Port, LED0_Pin, GPIO_PIN_RESET);
    HAL_GPIO_WritePin(LED1_GPIO_Port, LED1_Pin, GPIO_PIN_RESET);
    HAL_GPIO_WritePin(LED2_GPIO_Port, LED2_Pin, GPIO_PIN_RESET);

    // 再点亮当前的
    if (led_now == 0)
        HAL_GPIO_WritePin(LED0_GPIO_Port, LED0_Pin, GPIO_PIN_SET);
    else if (led_now == 1)
        HAL_GPIO_WritePin(LED1_GPIO_Port, LED1_Pin, GPIO_PIN_SET);
    else
        HAL_GPIO_WritePin(LED2_GPIO_Port, LED2_Pin, GPIO_PIN_SET);
}

// 刷新 OLED 显示（把数字转成字符串显示）
void Oled_Refresh(void)
{
    char buf[20];
    OLED_Clear();
    
    // 显示 STATE
    if (is_running == 1)
        OLED_ShowString(0, 0, "STATE: RUN  ");
    else
        OLED_ShowString(0, 0, "STATE: PAUSE");

    // 显示 DIR
    if (direction == 0)
        OLED_ShowString(0, 2, "DIR:   FWD  ");
    else
        OLED_ShowString(0, 2, "DIR:   REV  ");

    // 显示 STEP
    snprintf(buf, sizeof(buf), "STEP:  %dms", step_ms);
    OLED_ShowString(0, 4, buf);

    // 显示当前 LED 编号
    snprintf(buf, sizeof(buf), "LED:   %d", led_now + 1);
    OLED_ShowString(0, 6, buf);
}

// 串口反馈（发送当前状态，阻塞发送）
void Uart_Send_Feedback(void)
{
    char tx_buf[64];
    int len = snprintf(tx_buf, sizeof(tx_buf),
                       "STATE=%s,DIR=%s,STEP=%d,LED=%d\r\n",
                       is_running ? "RUN" : "PAUSE",
                       direction ? "REV" : "FWD",
                       step_ms,
                       led_now + 1);
    HAL_UART_Transmit(&huart2, (uint8_t*)tx_buf, len, 100);
}

// 处理串口收到的数据包（校验和格式：AA run dir stepL stepH sum）
void Process_Rx_Packet(void)
{
    // 1. 校验帧头和校验和
    uint8_t sum = rx_buf[0] + rx_buf[1] + rx_buf[2] + rx_buf[3] + rx_buf[4];
    if (rx_buf[0] != 0xAA || sum != rx_buf[5])
        return;     // 帧头不对或校验错，直接丢弃

    // 2. 把两个单字节拼成一个 16 位 step 值
    uint16_t new_step = (uint16_t)(rx_buf[3] | (rx_buf[4] << 8));

    // 3. 范围判断（100 ~ 2000 之间才合法）
    if (new_step < 100 || new_step > 2000)
        return;

    // 4. 更新状态
    is_running = rx_buf[1] ? 1 : 0;
    direction  = rx_buf[2] ? 1 : 0;
    step_ms    = new_step;
    oled_need_update = 1;       // 状态变了，通知刷新
}

// 2. 定时器中断回调（1ms 进一次）
void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)

    if (htim->Instance == TIM2)
    {
        tick_count++;
    }
}

// 3. 串口接收完成回调（收 1 字节进一次）
void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
    if (huart->Instance == USART1)
    {
        // 【关键修复】如果是第一字节，必须是 0xAA，否则丢弃并重置
        if (rx_cnt == 0)
        {
            if (rx_byte != 0xAA)
            {
                // 不是帧头，直接忽略，不存入缓冲区，保持 rx_cnt=0
                HAL_UART_Receive_IT(&huart2, &rx_byte, 1);
                return;
            }
        }

        rx_buf[rx_cnt] = rx_byte;   // 存进缓冲区
        rx_cnt++;

        if (rx_cnt >= 6)            // 收满一帧
        {
            rx_cnt = 0;             // 重置计数器，准备下一帧
            rx_packet_ready = 1;    // 通知主循环处理
        }

        // 必须重新开启接收中断，否则只能收一次！
        HAL_UART_Receive_IT(&huart2, &rx_byte, 1);
    }
}
